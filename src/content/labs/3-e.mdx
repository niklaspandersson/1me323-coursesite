Det fungerar för några få laborationer, men det blir snabbt mycket repetition och risken att något blir fel någonstans är stor. Öppna din `labs`-mapp i VS Code. Du ska arbeta i roten av denna mapp (där din `index.html` ligger).

### 1 - Problemet med hårdkodad HTML
Förra veckan skapade du en landing page med hårdkodad HTML för varje labb. Detta fungerar för få laborationer, men blir snabbt problem.

**Exempel på problemet:**
```html
<!-- Hårdkodad HTML - svår att underhålla -->
<ul>
  <li class="lab-card">
    <h3>Labb 1a - Moduler</h3>
    <p>Lär dig om import och export i JavaScript</p>
    <span class="status completed">Klar</span>
  </li>
  <li class="lab-card">
    <h3>Labb 1b - Git grunder</h3>
    <p>Versionshantering med Git och GitHub</p>
    <span class="status in-progress">Pågående</span>
  </li>
  <!-- ... 20+ laborationer mer? -->
</ul>
```

**Problem med detta:**
- Mycket repetition (varje labb kräver samma HTML-struktur)
- Risk för slarvfel och inkonsistens
- Svårt att ändra utseende på alla labbar samtidigt
- Svårt att lägga till ny funktionalitet (t.ex. filtrering)

#### Övning
1. Reflektera över din nuvarande landing page:

   - Hur många rader HTML är det per labb?
   - Vad skulle hända om du ville lägga till en "favorit"-stjärna på varje labb?

**Den datadrivna lösningen**: Istället för att hårdkoda HTML beskriver vi laborationerna som data och genererar HTML med JavaScript.

### 4 - Generera HTML från data med createElement
Nu kommer kärnan: vi ska skriva JavaScript som omvandlar vår data till HTML. Vi använder `document.createElement()` och `append()` som du lärde dig i förra laborationen.

**Exempel på renderingsfunktion med createElement:**
```javascript
function getStatusText(status) {
  const statusMap = {
    'completed': 'Klar',
    'in-progress': 'Pågående',
    'not-started': 'Ej startad'
  };
  return statusMap[status] || status;
}

function createLabCard(lab) {
  // Skapa huvudcontainern
  const card = document.createElement('div');
  card.classList.add('lab-card');

  // Skapa rubrik
  const heading = document.createElement('h3');
  heading.textContent = `Labb ${lab.id} - ${lab.title}`;

  // Skapa beskrivning
  const description = document.createElement('p');
  description.textContent = lab.description;

  // Skapa meta-container
  const meta = document.createElement('div');
  meta.classList.add('lab-meta');

  // Skapa status-span
  const statusSpan = document.createElement('span');
  statusSpan.classList.add('status', lab.status);
  statusSpan.textContent = getStatusText(lab.status);

  // Skapa vecka-span
  const weekSpan = document.createElement('span');
  weekSpan.classList.add('week');
  weekSpan.textContent = `Vecka ${lab.week}`;

  meta.append(statusSpan, weekSpan);

  // Lägg till länk om den finns
  if (lab.hasLink) {
    const link = document.createElement('a');
    link.href = `./${lab.id}/`;
    link.classList.add('lab-link');
    link.textContent = 'Öppna labb';
    card.append(heading, description, meta, link);
  } else {
    card.append(heading, description, meta);
  }

  return card;
}

function renderLabs(labsToShow = labs) {
  const container = document.getElementById('labs-container');

  // Rensa befintligt innehåll
  container.innerHTML = '';

  // Skapa och lägg till kort för varje labb
  for (const lab of labsToShow) {
    const card = createLabCard(lab);
    container.append(card);
  }
}

// Rendera vid sidladdning
renderLabs();
```

### 5 - Lägg till interaktivitet
Nu när grundstrukturen fungerar ska vi lägga till funktionalitet för filtrering och sortering.

**Exempel på filtreringsfunktion:**
```javascript
// Filtrera labbar baserat på status
function filterLabs(statusFilter) {
  let filteredLabs;

  if (statusFilter === 'all') {
    filteredLabs = labs;
  } else {
    filteredLabs = labs.filter(lab => lab.status === statusFilter);
  }

  renderLabs(filteredLabs);
  updateActiveFilter(statusFilter);
}

function updateActiveFilter(activeFilter) {
  // Ta bort 'active' klass från alla knappar
  document.querySelectorAll('nav button').forEach(btn => {
    btn.classList.remove('active');
  });

  // Lägg till 'active' klass på rätt knapp
  document.getElementById(`filter-${activeFilter}`).classList.add('active');
}

// Lägg till event listeners för filter-knappar
document.getElementById('filter-all').addEventListener('click', () => filterLabs('all'));
document.getElementById('filter-completed').addEventListener('click', () => filterLabs('completed'));
document.getElementById('filter-in-progress').addEventListener('click', () => filterLabs('in-progress'));
```

```javascript
function sortLabs(sortBy) {
  let sortedLabs = [...labs]; // Skapa en kopia

  switch(sortBy) {
    case 'week':
      sortedLabs.sort((a, b) => a.week - b.week);
      break;
    case 'title':
      sortedLabs.sort((a, b) => a.title.localeCompare(b.title));
      break;
    case 'status':
      const statusOrder = { 'completed': 0, 'in-progress': 1, 'not-started': 2 };
      sortedLabs.sort((a, b) => statusOrder[a.status] - statusOrder[b.status]);
      break;
  }

  renderLabs(sortedLabs);
}

// Sökfunktion
function searchLabs(searchTerm) {
  const filtered = labs.filter(lab =>
    lab.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
    lab.description.toLowerCase().includes(searchTerm.toLowerCase())
  );
  renderLabs(filtered);
}

// Event listener för sökfält
document.getElementById('search').addEventListener('input', (e) => {
  searchLabs(e.target.value);
});
```

### 6 - Utmaning: Komplett portfolio med localStorage
Nu ska du kombinera allt du lärt dig för att skapa en riktigt smart portfolio.

#### Uppgift
Integrera localStorage för att spara och ladda statusen på dina laborationer:

**Funktionalitet att implementera:**
1. **Spara status**: När användaren klickar på en labb, växla status och spara till localStorage
2. **Ladda status**: Vid sidladdning, läs sparad status och uppdatera labs-arrayen
3. **Progress tracking**: Visa progression (t.ex. "7 av 15 laborationer klara")
4. **Senaste aktivitet**: Spara datum för när labbar markeras som klara

**Tekniska krav:**
- Använd säkra localStorage-funktioner (från lab 2-d)
- Hantera fallet när ingen data finns sparad
- Uppdatera visningen direkt när status ändras

#### Starthjälp
```javascript
// Spara lab-status till localStorage
function saveLabStatus() {
  const statusData = labs.map(lab => ({
    id: lab.id,
    status: lab.status,
    lastModified: new Date().toISOString()
  }));

  saveToStorage('labStatus', statusData);
}

// Ladda och applicera sparad status
function loadLabStatus() {
  const savedStatus = loadFromStorage('labStatus', []);

  // Uppdatera labs-arrayen med sparad status
  savedStatus.forEach(saved => {
    const lab = labs.find(l => l.id === saved.id);
    if (lab) {
      lab.status = saved.status;
      lab.lastModified = saved.lastModified;
    }
  });
}

// Växla status för en labb
function toggleLabStatus(labId) {
  const lab = labs.find(l => l.id === labId);
  if (!lab) return;

  // Cykel genom statusar
  const statusCycle = ['not-started', 'in-progress', 'completed'];
  const currentIndex = statusCycle.indexOf(lab.status);
  const nextIndex = (currentIndex + 1) % statusCycle.length;

  lab.status = statusCycle[nextIndex];
  lab.lastModified = new Date().toISOString();

  saveLabStatus();
  renderLabs();
  updateProgress();
}

// Visa progression med createElement
function updateProgress() {
  const completed = labs.filter(lab => lab.status === 'completed').length;
  const total = labs.length;
  const percentage = Math.round((completed / total) * 100);

  const progressContainer = document.getElementById('progress');
  progressContainer.innerHTML = ''; // Rensa först

  const text = document.createElement('p');
  text.textContent = `Progression: ${completed} av ${total} laborationer klara (${percentage}%)`;

  const progressBar = document.createElement('div');
  progressBar.classList.add('progress-bar');

  const progressFill = document.createElement('div');
  progressFill.classList.add('progress-fill');
  progressFill.style.width = `${percentage}%`;

  progressBar.append(progressFill);
  progressContainer.append(text, progressBar);
}
```
